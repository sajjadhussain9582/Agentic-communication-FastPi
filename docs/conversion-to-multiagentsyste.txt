class RAGTool(BaseModel):
    """Search knowledge base"""
    query: str

    def __call__(self):
        return searchsimilar(self.query)


class EstimatorTool(BaseModel):
    """Estimate project cost and timeline"""
    project_desc: str

    def __call__(self):
        return estimateproject(self.project_desc)


class CalendlyTool(BaseModel):
    """Generate booking link"""
    duration: Optional[str] = "30min"

    async def __call__(self):
        return await calendly.getbookingurl(self.duration)

## 🧠 STEP 4: SUPERVISOR ROUTER

```python
def route_after_classify(state: AgentState):
    classification = state.get("classification")

    if not classification:
        return "qualifier"

    if classification.isescalated:
        return "humanreview"

    if classification.intenttype in ["pricingrequest", "quote", "budget"]:
        return "estimator"

    if classification.newpipelinestage in ["qualified", "meetingbooked"]:
        return "drafter"

    return "qualifier"
```

## 🔁 STEP 5: GRAPH BUILD

```python
def buildgraph(session: Session):
    g = StateGraph(AgentState)

    # Existing nodes
    g.add_node("classify", node_classify)
    g.add_node("updatestatus", node_update_status)
    g.add_node("humanreview", nodehumanreview)

    # Agents
    g.add_node("qualifier", qualifier)
    g.add_node("estimator", estimator)
    g.add_node("drafter", drafter)

    # Tool nodes (separate per agent)
    g.add_node("qualifier_tools", ToolNode([RAGTool()]))
    g.add_node("estimator_tools", ToolNode([EstimatorTool()]))
    g.add_node("drafter_tools", ToolNode([CalendlyTool()]))

    # Entry
    g.set_entry_point("classify")

    # Routing
    g.add_conditional_edges(
        "classify",
        route_after_classify,
        {
            "qualifier": "qualifier",
            "estimator": "estimator",
            "drafter": "drafter",
            "humanreview": "humanreview"
        }
    )

    # Tool loops
    g.add_conditional_edges("qualifier", tools_condition)
    g.add_edge("qualifier_tools", "qualifier")

    g.add_conditional_edges("estimator", tools_condition)
    g.add_edge("estimator_tools", "estimator")

    g.add_conditional_edges("drafter", tools_condition)
    g.add_edge("drafter_tools", "drafter")

    # Flow chaining
    g.add_edge("qualifier", "estimator")
    g.add_edge("estimator", "drafter")

    # Final
    g.add_edge("drafter", "updatestatus")
    g.add_edge("updatestatus", END)

    return g.compile(
        checkpointer=getcheckpointer(),
        interrupt_before=["humanreview"]
    )
```

## 🌊 STEP 6: STREAMING (Next.js UI)

```python
async def runaipipeline(...):
    graph = buildgraph(session)

    config = {
        "configurable": {
            "thread_id": str(conversation.id)
        }
    }

    async for event in graph.astream_events(initial_state, config, version="v2"):
        if event["event"] == "on_chat_model_stream":
            chunk = event["data"]["chunk"]
            if chunk.content:
                yield chunk.content

    final_state = await graph.aget_state(config)

    return final_state
```
